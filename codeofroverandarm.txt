// =====================================================
// STILL HERE RESCUE ROVER
// Controller: ESP32-S3
// Rover driver: HW-095 / L298N-style
// Arm driver: PCA9685
// Sensors: Left IR, Right IR, MQ-type gas, flame
// =====================================================


// ---------------- WIFI SETTINGS ----------------

const char* AP_SSID = "STILL_HERE_ROVER";
const char* AP_PASSWORD = "Rescue123";  // Change this password


// ---------------- HW-095 MOTOR PINS ----------------

// Left-side motors
const int ENA = 6;
const int IN1 = 4;
const int IN2 = 5;

// Right-side motors
const int ENB = 9;
const int IN3 = 7;
const int IN4 = 8;


// ---------------- PCA9685 I2C PINS ----------------

const int SDA_PIN = 11;
const int SCL_PIN = 12;

const uint8_t PCA9685_ADDRESS = 0x40;

Adafruit_PWMServoDriver pwm(PCA9685_ADDRESS);
bool pcaReady = false;


// ---------------- SENSOR PINS ----------------

const int IR_LEFT_PIN  = 13;
const int IR_RIGHT_PIN = 14;

const int FLAME_DO_PIN = 15;

const int GAS_DO_PIN = 16;  // Optional digital output
const int GAS_AO_PIN = 1;   // Analog output through a divider


// ---------------- SENSOR SETTINGS ----------------

// Many IR obstacle modules output LOW when an
// obstacle is detected. Change these if needed.
const bool IR_ACTIVE_LOW = true;

// Many flame modules output LOW when flame is detected.
const bool FLAME_ACTIVE_LOW = true;

// Many MQ-module digital comparator outputs go LOW
// when their adjustable threshold is crossed.
const bool GAS_DO_ACTIVE_LOW = true;

// Set true if the analog gas reading normally increases
// when the sensor responds to the gas of interest.
const bool GAS_HIGH_READING_MEANS_ALERT = true;

// This is a starting experimental ADC threshold, not
// a calibrated concentration or a certified alarm level.
const int GAS_THRESHOLD = 2500;


// ---------------- SAFETY SETTINGS ----------------

// IR interlock: prevent moving FORWARD when either
// front IR sensor reports an obstacle.
const bool ENABLE_IR_FORWARD_STOP = true;

// Initially disabled until sensor outputs are checked
// on the webpage and their polarity is verified.
const bool ENABLE_FLAME_STOP = false;
const bool ENABLE_GAS_ANALOG_STOP = false;
const bool ENABLE_GAS_DIGITAL_STOP = false;


// ---------------- SERVO SETTINGS ----------------

// These conservative limits are only starting values.
// Adjust each channel for your actual arm mechanism.
const int SERVO_MIN_DEG[4] = {20, 20, 20, 20};
const int SERVO_MAX_DEG[4] = {160, 160, 160, 160};

const int SERVO_START_DEG = 90;


// ---------------- MOTOR / NETWORK SETTINGS ----------------

const unsigned long DRIVE_TIMEOUT_MS = 450;

WebServer server(80);

bool driveActive = false;
unsigned long lastDriveCommand = 0;

String lastBlockReason = "";


// =====================================================
// SENSOR READING FUNCTIONS
// =====================================================

bool readActiveLevel(int pin, bool activeLow) {
  int value = digitalRead(pin);

  if (activeLow) {
    return value == LOW;
  }

  return value == HIGH;
}


bool irLeftDetected() {
  return readActiveLevel(IR_LEFT_PIN, IR_ACTIVE_LOW);
}


bool irRightDetected() {
  return readActiveLevel(IR_RIGHT_PIN, IR_ACTIVE_LOW);
}


bool flameDetected() {
  return readActiveLevel(FLAME_DO_PIN, FLAME_ACTIVE_LOW);
}


bool gasDigitalDetected() {
  return readActiveLevel(GAS_DO_PIN, GAS_DO_ACTIVE_LOW);
}


int readGasAnalog() {
  return analogRead(GAS_AO_PIN);
}


bool gasAnalogThresholdCrossed(int rawValue) {
  if (GAS_HIGH_READING_MEANS_ALERT) {
    return rawValue >= GAS_THRESHOLD;
  }

  return rawValue <= GAS_THRESHOLD;
}


String jsonBool(bool value) {
  return value ? "true" : "false";
}


// =====================================================
// MOTOR CONTROL FUNCTIONS
// =====================================================

void setMotor(
  int input1,
  int input2,
  int enablePin,
  int speedValue
) {
  speedValue = constrain(speedValue, -255, 255);

  if (speedValue > 0) {
    digitalWrite(input1, HIGH);
    digitalWrite(input2, LOW);
  }
  else if (speedValue < 0) {
    digitalWrite(input1, LOW);
    digitalWrite(input2, HIGH);
  }
  else {
    digitalWrite(input1, LOW);
    digitalWrite(input2, LOW);
  }

  analogWrite(enablePin, abs(speedValue));
}


void stopMotors() {
  setMotor(IN1, IN2, ENA, 0);
  setMotor(IN3, IN4, ENB, 0);

  driveActive = false;
}


bool checkMovementSafety(char direction) {

  // Flame interlock can be enabled after verifying
  // the sensor's detection polarity.
  if (ENABLE_FLAME_STOP && flameDetected()) {
    lastBlockReason = "Flame sensor triggered";
    return false;
  }

  int gasRaw = readGasAnalog();

  if (ENABLE_GAS_ANALOG_STOP &&
      gasAnalogThresholdCrossed(gasRaw)) {
    lastBlockReason = "Gas analog threshold crossed";
    return false;
  }

  if (ENABLE_GAS_DIGITAL_STOP && gasDigitalDetected()) {
    lastBlockReason = "Gas digital threshold crossed";
    return false;
  }

  // Stop forward motion if either front IR sensor
  // identifies an obstacle.
  if (direction == 'F' &&
      ENABLE_IR_FORWARD_STOP &&
      (irLeftDetected() || irRightDetected())) {
    lastBlockReason = "Front obstacle detected";
    return false;
  }

  return true;
}


bool moveRover(char direction, int speedValue) {

  speedValue = constrain(speedValue, 0, 255);

  if (!(direction == 'F' ||
        direction == 'B' ||
        direction == 'L' ||
        direction == 'R')) {
    stopMotors();
    lastBlockReason = "Invalid movement command";
    return false;
  }

  if (!checkMovementSafety(direction)) {
    stopMotors();
    return false;
  }

  int leftSpeed = 0;
  int rightSpeed = 0;

  switch (direction) {

    case 'F':
      leftSpeed = speedValue;
      rightSpeed = speedValue;
      break;

    case 'B':
      leftSpeed = -speedValue;
      rightSpeed = -speedValue;
      break;

    case 'L':
      leftSpeed = -speedValue;
      rightSpeed = speedValue;
      break;

    case 'R':
      leftSpeed = speedValue;
      rightSpeed = -speedValue;
      break;
  }

  setMotor(IN1, IN2, ENA, leftSpeed);
  setMotor(IN3, IN4, ENB, rightSpeed);

  lastDriveCommand = millis();
  driveActive = true;

  lastBlockReason = "";

  return true;
}


// =====================================================
// PCA9685 SERVO CONTROL
// =====================================================

bool setArmServo(int channel, int angle) {

  if (!pcaReady) {
    return false;
  }

  if (channel < 0 || channel > 3) {
    return false;
  }

  angle = constrain(
    angle,
    SERVO_MIN_DEG[channel],
    SERVO_MAX_DEG[channel]
  );

  // Initial calibration range for typical hobby servos.
  // Adjust to the servo manufacturer's specification.
  const int pulseMinUs = 1000;
  const int pulseMaxUs = 2000;
  const int periodUs = 20000;  // 50 Hz

  int pulseUs = map(
    angle,
    0,
    180,
    pulseMinUs,
    pulseMaxUs
  );

  uint16_t ticks =
    ((uint32_t)pulseUs * 4096UL) / periodUs;

  pwm.setPWM(channel, 0, ticks);

  return true;
}

// =====================================================
// WEB SERVER HANDLERS
// =====================================================

void handleRoot() {
  server.send_P(200, "text/html", webpage);
}


void handleDrive() {

  if (!server.hasArg("dir")) {
    server.send(400, "text/plain", "Missing direction");
    return;
  }

  String directionText = server.arg("dir");

  if (directionText.length() == 0) {
    server.send(400, "text/plain", "Empty direction");
    return;
  }

  char direction = directionText.charAt(0);

  int speedValue = 120;

  if (server.hasArg("speed")) {
    speedValue = server.arg("speed").toInt();
  }

  speedValue = constrain(speedValue, 0, 255);

  if (!moveRover(direction, speedValue)) {
    server.send(409, "text/plain", lastBlockReason);
    return;
  }

  server.send(200, "text/plain", "Moving");
}


void handleStop() {
  stopMotors();
  lastBlockReason = "";

  server.send(200, "text/plain", "Stopped");
}


void handleServo() {

  if (!server.hasArg("ch") || !server.hasArg("angle")) {
    server.send(400, "text/plain", "Missing channel or angle");
    return;
  }

  if (!pcaReady) {
    server.send(503, "text/plain", "PCA9685 not detected");
    return;
  }

  int channel = server.arg("ch").toInt();
  int angle = server.arg("angle").toInt();

  if (channel < 0 || channel > 3) {
    server.send(400, "text/plain", "Channel must be 0 to 3");
    return;
  }

  angle = constrain(
    angle,
    SERVO_MIN_DEG[channel],
    SERVO_MAX_DEG[channel]
  );

  if (!setArmServo(channel, angle)) {
    server.send(500, "text/plain", "Servo command failed");
    return;
  }

  server.send(200, "text/plain", "Servo command accepted");
}


void handleStatus() {

  bool irL = irLeftDetected();
  bool irR = irRightDetected();
  bool flame = flameDetected();
  bool gasDO = gasDigitalDetected();

  int gasRaw = readGasAnalog();
  bool gasAlert = gasAnalogThresholdCrossed(gasRaw);

  String json = "{";

  json += "\"irLeft\":";
  json += jsonBool(irL);
  json += ",";

  json += "\"irRight\":";
  json += jsonBool(irR);
  json += ",";

  json += "\"flame\":";
  json += jsonBool(flame);
  json += ",";

  json += "\"gasRaw\":";
  json += String(gasRaw);
  json += ",";

  json += "\"gasAnalogAlert\":";
  json += jsonBool(gasAlert);
  json += ",";

  json += "\"gasDigital\":";
  json += jsonBool(gasDO);
  json += ",";

  json += "\"pcaReady\":";
  json += jsonBool(pcaReady);
  json += ",";

  json += "\"reason\":\"";
  json += lastBlockReason;
  json += "\"";

  json += "}";

  server.send(200, "application/json", json);
}


// =====================================================
// SETUP
// =====================================================

void setup() {

  Serial.begin(115200);
  delay(300);

  // Motor pins
  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(ENB, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  stopMotors();

  // Digital sensor pins
  pinMode(IR_LEFT_PIN, INPUT_PULLUP);
  pinMode(IR_RIGHT_PIN, INPUT_PULLUP);
  pinMode(FLAME_DO_PIN, INPUT_PULLUP);
  pinMode(GAS_DO_PIN, INPUT_PULLUP);

  // Gas sensor ADC
  analogReadResolution(12);
  analogSetPinAttenuation(GAS_AO_PIN, ADC_11db);

  // PCA9685
  Wire.begin(SDA_PIN, SCL_PIN);
  Wire.setClock(100000);

  Wire.beginTransmission(PCA9685_ADDRESS);

  if (Wire.endTransmission() == 0) {
    pwm.begin();
    pwm.setPWMFreq(50);

    pcaReady = true;

    Serial.println("PCA9685 detected and initialized.");

    // Hold the initial arm position at approximately 90 degrees.
    for (int ch = 0; ch < 4; ch++) {
      setArmServo(ch, SERVO_START_DEG);
    }
  } else {
    pcaReady = false;
    Serial.println("PCA9685 NOT detected. Check I2C wiring.");
  }

  // Wi-Fi access point
  WiFi.mode(WIFI_AP);
  WiFi.softAP(AP_SSID, AP_PASSWORD);

  Serial.println();
  Serial.println("STILL HERE Rover ready.");
  Serial.print("Wi-Fi name: ");
  Serial.println(AP_SSID);
  Serial.print("Open browser at: http://");
  Serial.println(WiFi.softAPIP());

  // Routes
  server.on("/", HTTP_GET, handleRoot);
  server.on("/drive", HTTP_GET, handleDrive);
  server.on("/stop", HTTP_GET, handleStop);
  server.on("/servo", HTTP_GET, handleServo);
  server.on("/status", HTTP_GET, handleStatus);

  server.begin();

  Serial.println("Web server started.");
}


// =====================================================
// MAIN LOOP
// =====================================================

void loop() {

  server.handleClient();

  // Stop the wheels if movement commands stop arriving.
  if (driveActive &&
      millis() - lastDriveCommand > DRIVE_TIMEOUT_MS) {

    stopMotors();
    lastBlockReason = "Movement command timeout";
  }
}
