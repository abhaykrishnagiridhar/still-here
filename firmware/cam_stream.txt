// REFERENCE ONLY, not built by PlatformIO (it is a .txt file on purpose).
// Recovered from an older copy of the project ("reference from chatgpt").
//
// What it is: a standalone camera stream for an AI-Thinker-pin ESP32-CAM board whose sensor is a
// GC2145, not the usual OV2640. The team now uses the ESP32-S3 board with its built-in camera
// socket, whose video comes from camera.cpp and mjpeg_server.cpp, so this is kept as the working
// example of the other route, and for what it learned:
//   - A GC2145 cannot make JPEG pictures itself. Capture RGB565 (small, QVGA) and convert each
//     picture with frame2jpg() on the ESP32, which is the reliable low-memory way.
//   - Do not use CAMERA_MODEL_AI_THINKER's defaults: they expect an OV2640. A recent Espressif
//     camera driver detects the GC2145 by itself; an old board package fails with ESP_ERR_NOT_SUPPORTED,
//     so update "esp32 by Espressif Systems" in the Arduino IDE first.
//   - It needs PSRAM, and 2.4 GHz Wi-Fi.
//
// To use it: in the Arduino IDE make a new sketch, paste this in, and add a tab named secrets.h
// (copy firmware/secrets.example.h and fill in your Wi-Fi; never commit it). Select the board
// "AI Thinker ESP32-CAM", enable PSRAM, upload, and open the address the Serial Monitor prints
// (115200 baud). It serves / (a page), /stream (MJPEG), /capture (one picture) and /health.
// It serves the stream on port 80, so keep it on its own board, apart from the rover firmware.

// GC2145 camera stream for an AI-Thinker-pin-compatible ESP32-CAM.
//
// The GC2145 does not create JPEG frames itself. Capture small RGB565 frames
// and encode them to JPEG before placing them in the browser MJPEG stream.
// Keep this firmware separate from the rover ESP32-S3 firmware.

#include <Arduino.h>
#include <WiFi.h>
#include <WebServer.h>

#include "esp_camera.h"
#include "img_converters.h"
#include "secrets.h"

namespace {

// AI-Thinker ESP32-CAM camera bus. These pins are fixed by the PCB layout,
// including on common GC2145 / M21-45 camera-module replacements.
constexpr int PIN_PWDN = 32;
constexpr int PIN_RESET = -1;
constexpr int PIN_XCLK = 0;
constexpr int PIN_SIOD = 26;
constexpr int PIN_SIOC = 27;
constexpr int PIN_D7 = 35;
constexpr int PIN_D6 = 34;
constexpr int PIN_D5 = 39;
constexpr int PIN_D4 = 36;
constexpr int PIN_D3 = 21;
constexpr int PIN_D2 = 19;
constexpr int PIN_D1 = 18;
constexpr int PIN_D0 = 5;
constexpr int PIN_VSYNC = 25;
constexpr int PIN_HREF = 23;
constexpr int PIN_PCLK = 22;

constexpr char STREAM_BOUNDARY[] = "frame";
WebServer server(80);

const char INDEX_HTML[] PROGMEM = R"HTML(
<!doctype html><html><head><meta name="viewport" content="width=device-width,initial-scale=1">
<title>STILL HERE camera</title><style>body{margin:0;background:#111;color:#fff;font:16px system-ui;text-align:center}main{max-width:800px;margin:auto;padding:16px}img{width:100%;height:auto;background:#222;border-radius:10px}a{color:#80e6ff}</style>
</head><body><main><h1>STILL HERE camera</h1><img src="/stream" alt="Live GC2145 camera stream"><p><a href="/capture">Save one frame</a> · <a href="/health">Health</a></p></main></body></html>
)HTML";

camera_config_t makeCameraConfig() {
  camera_config_t config = {};
  config.pin_pwdn = PIN_PWDN;
  config.pin_reset = PIN_RESET;
  config.pin_xclk = PIN_XCLK;
  config.pin_sccb_sda = PIN_SIOD;
  config.pin_sccb_scl = PIN_SIOC;
  config.pin_d7 = PIN_D7;
  config.pin_d6 = PIN_D6;
  config.pin_d5 = PIN_D5;
  config.pin_d4 = PIN_D4;
  config.pin_d3 = PIN_D3;
  config.pin_d2 = PIN_D2;
  config.pin_d1 = PIN_D1;
  config.pin_d0 = PIN_D0;
  config.pin_vsync = PIN_VSYNC;
  config.pin_href = PIN_HREF;
  config.pin_pclk = PIN_PCLK;
  config.xclk_freq_hz = 20'000'000;
  config.ledc_timer = LEDC_TIMER_0;
  config.ledc_channel = LEDC_CHANNEL_0;
  config.pixel_format = PIXFORMAT_RGB565;
  config.frame_size = FRAMESIZE_QVGA;
  config.jpeg_quality = 12;
  config.fb_count = 1;
  return config;
}

bool writeJpegFrame(WiFiClient &client, camera_fb_t *frame) {
  uint8_t *jpeg = frame->buf;
  size_t jpegLength = frame->len;
  bool ownsJpeg = false;

  if (frame->format != PIXFORMAT_JPEG) {
    if (!frame2jpg(frame, 60, &jpeg, &jpegLength)) {
      Serial.println("JPEG conversion failed");
      return false;
    }
    ownsJpeg = true;
  }

  const bool headerWritten = client.printf(
      "--%s\r\nContent-Type: image/jpeg\r\nContent-Length: %u\r\n\r\n",
      STREAM_BOUNDARY, static_cast<unsigned>(jpegLength));
  const bool frameWritten = headerWritten && client.write(jpeg, jpegLength) == jpegLength;
  const bool endingWritten = frameWritten && client.print("\r\n");

  if (ownsJpeg) {
    free(jpeg);
  }
  return endingWritten;
}

void handleCapture() {
  camera_fb_t *frame = esp_camera_fb_get();
  if (frame == nullptr) {
    server.send(503, "text/plain", "Camera frame unavailable");
    return;
  }

  uint8_t *jpeg = frame->buf;
  size_t jpegLength = frame->len;
  const bool alreadyJpeg = frame->format == PIXFORMAT_JPEG;
  const bool converted = alreadyJpeg || frame2jpg(frame, 60, &jpeg, &jpegLength);
  esp_camera_fb_return(frame);

  if (!converted) {
    server.send(500, "text/plain", "JPEG conversion failed");
    return;
  }

  WiFiClient client = server.client();
  client.printf("HTTP/1.1 200 OK\r\nContent-Type: image/jpeg\r\nContent-Length: %u\r\n"
                "Content-Disposition: inline; filename=gc2145.jpg\r\nConnection: close\r\n\r\n",
                static_cast<unsigned>(jpegLength));
  client.write(jpeg, jpegLength);
  if (!alreadyJpeg) {
    free(jpeg);
  }
}

void handleStream() {
  WiFiClient client = server.client();
  server.setContentLength(CONTENT_LENGTH_UNKNOWN);
  server.sendHeader("Cache-Control", "no-store");
  server.send(200, "multipart/x-mixed-replace; boundary=frame", "");

  while (client.connected()) {
    camera_fb_t *frame = esp_camera_fb_get();
    if (frame == nullptr) {
      Serial.println("Camera frame unavailable");
      break;
    }
    const bool wroteFrame = writeJpegFrame(client, frame);
    esp_camera_fb_return(frame);
    if (!wroteFrame) {
      break;
    }
    delay(1);
  }
}

void startWiFi() {
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.printf("Joining Wi-Fi SSID '%s'", WIFI_SSID);
  const unsigned long deadline = millis() + 30'000;
  while (WiFi.status() != WL_CONNECTED && millis() < deadline) {
    delay(250);
    Serial.print('.');
  }
  Serial.println();
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Wi-Fi connection failed. Check firmware/secrets.h and use 2.4 GHz Wi-Fi.");
    return;
  }
  Serial.printf("Camera ready: http://%s/\n", WiFi.localIP().toString().c_str());
}

}  // namespace

void setup() {
  Serial.begin(115200);
  Serial.setDebugOutput(false);
  delay(500);
  Serial.println("\nSTILL HERE GC2145 camera stream boot");

  if (!psramFound()) {
    Serial.println("PSRAM not found. GC2145 streaming requires PSRAM; check the board and Tools settings.");
  }

  camera_config_t cameraConfig = makeCameraConfig();
  const esp_err_t result = esp_camera_init(&cameraConfig);
  if (result != ESP_OK) {
    Serial.printf("GC2145 camera init failed: 0x%x\n", result);
    Serial.println("Install/update the Espressif ESP32 board package with GC2145 support, then reseat the camera ribbon.");
    return;
  }

  sensor_t *sensor = esp_camera_sensor_get();
  Serial.printf("Camera detected. Sensor PID: 0x%04x (expected GC2145)\n", sensor->id.PID);
  startWiFi();

  server.on("/", HTTP_GET, [] { server.send_P(200, "text/html", INDEX_HTML); });
  server.on("/capture", HTTP_GET, handleCapture);
  server.on("/stream", HTTP_GET, handleStream);
  server.on("/health", HTTP_GET, [] {
    server.send(200, "application/json", "{\"camera\":\"GC2145\",\"ready\":true}");
  });
  server.begin();
  Serial.println("HTTP server started");
}

void loop() {
  server.handleClient();
}
